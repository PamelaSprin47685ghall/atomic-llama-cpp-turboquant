cmake_minimum_required(VERSION 3.16)
project(TQCorpusCompose VERSION 0.1.0 LANGUAGES CXX)
include(GNUInstallDirs)
include(CTest)
if(NOT TARGET llama)
    find_package(llama CONFIG REQUIRED)
endif()
if(NOT TARGET TQCorpus::corpus)
    find_package(TQCorpus CONFIG REQUIRED)
endif()
add_executable(llama-corpus src/main.cpp)
target_compile_features(llama-corpus PRIVATE cxx_std_17)
target_link_libraries(llama-corpus PRIVATE llama TQCorpus::corpus)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(llama-corpus PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
set_target_properties(llama-corpus PROPERTIES INSTALL_RPATH "$ORIGIN/../${CMAKE_INSTALL_LIBDIR}")
if(BUILD_TESTING)
    add_test(NAME tq-corpus-help COMMAND llama-corpus --help)
endif()
install(TARGETS llama-corpus RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(FILES LICENSE DESTINATION ${CMAKE_INSTALL_DATADIR}/licenses/TQCorpusCompose)
