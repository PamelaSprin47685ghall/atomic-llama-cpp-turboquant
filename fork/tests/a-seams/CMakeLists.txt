# This explicitly opted-in test exercises an internal A backend capability.
# Its private header dependency does not belong to any installed C consumer.
option(TQ_BUILD_HIP_RELAY_PROVIDER_TEST "Build the real five-GPU HIP relay provider regression" OFF)
set(TQ_A_SOURCE_DIR "" CACHE PATH "Matching A source tree for the internal HIP relay provider test")
if(TQ_BUILD_HIP_RELAY_PROVIDER_TEST)
    if(NOT EXISTS "${TQ_A_SOURCE_DIR}/ggml/src/ggml-relay-impl.h")
        message(FATAL_ERROR "TQ_BUILD_HIP_RELAY_PROVIDER_TEST requires a matching TQ_A_SOURCE_DIR")
    endif()
    find_package(Threads REQUIRED)
    add_executable(test-hip-relay-rebinding hip-relay-rebinding.cpp)
    target_compile_features(test-hip-relay-rebinding PRIVATE cxx_std_17)
    target_include_directories(test-hip-relay-rebinding PRIVATE "${TQ_A_SOURCE_DIR}/ggml/src")
    target_link_libraries(test-hip-relay-rebinding PRIVATE llama ggml::ggml Threads::Threads)
    if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
        target_compile_options(test-hip-relay-rebinding PRIVATE -fno-fast-math -ffp-contract=off)
    endif()
    add_test(NAME hip-relay-rebinding COMMAND test-hip-relay-rebinding --timeout-seconds 120)
    set_tests_properties(hip-relay-rebinding PROPERTIES TIMEOUT 150 RUN_SERIAL TRUE)
endif()
