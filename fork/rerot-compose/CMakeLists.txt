cmake_minimum_required(VERSION 3.16)
project(TQRerotCompose VERSION 0.1.0 LANGUAGES CXX)

include(GNUInstallDirs)
find_package(llama CONFIG REQUIRED)
find_package(TQRerot CONFIG REQUIRED)
find_package(TQRelay CONFIG REQUIRED)
find_package(TQText CONFIG REQUIRED)
find_package(TQSampling CONFIG REQUIRED)

add_executable(tq_rerot_compose src/main.cpp src/native.cpp)
set_target_properties(tq_rerot_compose PROPERTIES OUTPUT_NAME tq-rerot-compose)
target_compile_features(tq_rerot_compose PRIVATE cxx_std_17)
target_link_libraries(tq_rerot_compose PRIVATE llama TQRerot::rerot TQRelay::relay TQText::text TQSampling::sampling)
if(UNIX AND NOT APPLE)
    set_target_properties(tq_rerot_compose PROPERTIES INSTALL_RPATH "$ORIGIN/../${CMAKE_INSTALL_LIBDIR}")
endif()
install(TARGETS tq_rerot_compose RUNTIME DESTINATION "${CMAKE_INSTALL_BINDIR}")
install(FILES README.md QUALIFICATION.md DESTINATION "${CMAKE_INSTALL_DATADIR}/doc/TQRerotCompose")
install(FILES LICENSE DESTINATION "${CMAKE_INSTALL_DATADIR}/licenses/TQRerotCompose")

include(CTest)
if(BUILD_TESTING)
    add_executable(test_rerot_compose_sampling tests/test-sampling.cpp)
    target_compile_features(test_rerot_compose_sampling PRIVATE cxx_std_17)
    target_include_directories(test_rerot_compose_sampling PRIVATE src)
    target_link_libraries(test_rerot_compose_sampling PRIVATE llama TQSampling::sampling)
    add_test(NAME tq_rerot_compose.sampling COMMAND test_rerot_compose_sampling)
    add_executable(test_rerot_compose_capacity tests/test-capacity.cpp)
    target_compile_features(test_rerot_compose_capacity PRIVATE cxx_std_17)
    target_include_directories(test_rerot_compose_capacity PRIVATE src)
    add_test(NAME tq_rerot_compose.capacity COMMAND test_rerot_compose_capacity)
    add_executable(test_rerot_compose_cancellation tests/test-native-cancellation.cpp src/native.cpp)
    target_compile_features(test_rerot_compose_cancellation PRIVATE cxx_std_17)
    target_include_directories(test_rerot_compose_cancellation PRIVATE src)
    target_link_libraries(test_rerot_compose_cancellation PRIVATE llama TQRerot::rerot TQRelay::relay)
    set(TQ_REROT_CANCEL_TEST_MODEL "" CACHE FILEPATH "Real model for native CPU cancellation checks")
    if(TQ_REROT_CANCEL_TEST_MODEL)
        add_test(NAME tq_rerot_compose.cancellation
            COMMAND test_rerot_compose_cancellation "${TQ_REROT_CANCEL_TEST_MODEL}")
        set_tests_properties(tq_rerot_compose.cancellation PROPERTIES TIMEOUT 120)
    endif()
    find_package(Python3 REQUIRED COMPONENTS Interpreter)
    add_test(NAME tq_rerot_compose.whiteboard_validator
        COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/test-whiteboard-validator.py")
    add_test(NAME tq_rerot_compose.cli
        COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_cli.py"
                --binary "$<TARGET_FILE:tq_rerot_compose>")
endif()
